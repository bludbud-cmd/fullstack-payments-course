This is the start of PayDeck itself. Everything from Part 3 comes together: components, layout, forms, navigation and saved data. In Part 7 the sample data is replaced by the real API; until then, the app works entirely on the device.

## What it does

- **Wallet** (first tab): the balance, how many cards are active, the three most recent transactions, and a button to see all activity.
- **Cards**: every card that is not cancelled, drawn with your payment card component. Tapping one opens its settings.
- **Card settings** (opened from Cards): the card, a **Freeze card** / **Unfreeze card** button, and a **Cancel card** button that cancels it and goes back.
- **Activity**: every transaction, newest first.
- **You**: a placeholder until sign-in arrives in Part 8.
- A tab bar on narrow screens, a sidebar on wide ones, and the Android back button working.
- Freezing or cancelling a card survives quitting and reopening the app.

```ts
export function formatMoney(cents: number): string {
  return new Intl.NumberFormat('en', { style: 'currency', currency: 'USD' }).format(cents / 100);
}

export function formatSignedMoney(cents: number): string {
  return (cents < 0 ? '−' : '+') + formatMoney(Math.abs(cents));
}
```

```ts
/** Every screen in the app, and what each one needs to know. */
export type Route =
  | { name: 'home' }
  | { name: 'cards' }
  | { name: 'activity' }
  | { name: 'profile' }
  | { name: 'card'; cardId: string };

/** The screens in the tab bar (phone) or sidebar (Mac). */
export const tabs = [
  { name: 'home', label: 'Wallet' },
  { name: 'cards', label: 'Cards' },
  { name: 'activity', label: 'Activity' },
  { name: 'profile', label: 'You' },
] as const;

export type TabName = (typeof tabs)[number]['name'];

/** A stack of routes: the last one is on screen. */
export type NavState = Route[];

export function push(stack: NavState, route: Route): NavState {
  return [...stack, route];
}

export function pop(stack: NavState): NavState {
  return stack.length > 1 ? stack.slice(0, -1) : stack;
}

export function switchTab(tab: TabName): NavState {
  return [{ name: tab }];
}

export function current(stack: NavState): Route {
  return stack[stack.length - 1];
}
```

```tsx
import { createContext, useContext } from 'react';
import type { Route, TabName } from './routes';

export type Navigation = {
  navigate: (route: Route) => void;
  goBack: () => void;
  switchTab: (tab: TabName) => void;
  canGoBack: boolean;
};

export const NavigationContext = createContext<Navigation | null>(null);

export function useNavigation(): Navigation {
  const navigation = useContext(NavigationContext);
  if (navigation === null) {
    throw new Error('useNavigation must be used inside <AppNavigator>');
  }
  return navigation;
}
```

`src/navigation/AppNavigator.tsx`:

```tsx
import { useEffect, useState } from 'react';
import { BackHandler, Pressable, StyleSheet, Text, useWindowDimensions, View } from 'react-native';
import { ActivityScreen } from '../screens/ActivityScreen';
import { CardDetailScreen } from '../screens/CardDetailScreen';
import { CardsScreen } from '../screens/CardsScreen';
import { HomeScreen } from '../screens/HomeScreen';
import { ProfileScreen } from '../screens/ProfileScreen';
import { NavigationContext, type Navigation } from './NavigationContext';
import { current, pop, push, switchTab, tabs, type NavState, type Route } from './routes';

const WIDE_LAYOUT = 720;

function renderScreen(route: Route) {
  switch (route.name) {
    case 'home':
      return <HomeScreen />;
    case 'cards':
      return <CardsScreen />;
    case 'activity':
      return <ActivityScreen />;
    case 'profile':
      return <ProfileScreen />;
    case 'card':
      return <CardDetailScreen cardId={route.cardId} />;
  }
}

export function AppNavigator() {
  const { width } = useWindowDimensions();
  const wide = width >= WIDE_LAYOUT;
  const [stack, setStack] = useState<NavState>([{ name: 'home' }]);

  const navigation: Navigation = {
    navigate: (route) => setStack(push(stack, route)),
    goBack: () => setStack(pop(stack)),
    switchTab: (tab) => setStack(switchTab(tab)),
    canGoBack: stack.length > 1,
  };

  // The Android back button goes back a screen instead of closing the app.
  useEffect(() => {
    const subscription = BackHandler.addEventListener('hardwareBackPress', () => {
      if (stack.length > 1) {
        setStack(pop(stack));
        return true;
      }
      return false;
    });
    return () => subscription.remove();
  }, [stack]);

  const activeTab = stack[0].name;
  const tabButtons = tabs.map((tab) => {
    const selected = tab.name === activeTab;
    return (
      <Pressable
        key={tab.name}
        accessibilityRole="tab"
        accessibilityState={{ selected }}
        onPress={() => navigation.switchTab(tab.name)}
        style={[wide ? styles.sidebarItem : styles.tabItem, selected && wide && styles.sidebarSelected]}
      >
        <Text style={[styles.tabLabel, selected && styles.tabLabelSelected]}>{tab.label}</Text>
      </Pressable>
    );
  });

  return (
    <NavigationContext.Provider value={navigation}>
      <View style={[styles.root, wide && styles.rootWide]}>
        {wide && (
          <View accessibilityRole="tablist" style={styles.sidebar}>
            <Text style={styles.brand}>PayDeck</Text>
            {tabButtons}
          </View>
        )}
        <View style={styles.content}>{renderScreen(current(stack))}</View>
        {!wide && (
          <View accessibilityRole="tablist" style={styles.tabBar}>
            {tabButtons}
          </View>
        )}
      </View>
    </NavigationContext.Provider>
  );
}

const styles = StyleSheet.create({
  root: { flex: 1, backgroundColor: '#f5f6fa' },
  rootWide: { flexDirection: 'row' },
  content: { flex: 1 },
  sidebar: {
    width: 200,
    paddingTop: 24,
    paddingHorizontal: 8,
    gap: 2,
    backgroundColor: '#ffffff',
    borderRightWidth: 1,
    borderRightColor: '#dfe1ea',
  },
  brand: { fontSize: 20, fontWeight: '800', paddingHorizontal: 12, marginBottom: 16 },
  sidebarItem: { paddingVertical: 10, paddingHorizontal: 12, borderRadius: 8 },
  sidebarSelected: { backgroundColor: '#f5f6fa' },
  tabBar: {
    flexDirection: 'row',

    borderTopWidth: 1,

  tabLabel: { fontSize: 15, color: '#5d6177' },
  tabLabelSelected: { color: '#3c4fe0', fontWeight: '700' },
});
```

`src/state/useStoredState.ts`:

```ts
import AsyncStorage from '@react-native-async-storage/async-storage';
import { useEffect, useState } from 'react';

/**
 * Like useState, but the value is saved on the device under `key`
 * and loaded again next time the app starts.
 */
export function useStoredState<T>(key: string, initial: T) {
  const [value, setValue] = useState<T>(initial);
  const [loaded, setLoaded] = useState(false);

  // Load what was saved last time, once.
  useEffect(() => {
    AsyncStorage.getItem(key)
      .then((saved) => {
        if (saved !== null) {
          setValue(JSON.parse(saved));
        }
      })
      .catch(() => {
        // Nothing could be read: keep the starting value.
      })
      .finally(() => setLoaded(true));
  }, [key]);

  // Save after every change, once loading has finished.
  useEffect(() => {
    if (loaded) {
      AsyncStorage.setItem(key, JSON.stringify(value)).catch(() => {});
    }
  }, [key, value, loaded]);

  return [value, setValue, loaded] as const;
}
```

`src/state/WalletContext.tsx`:

```tsx
import { createContext, useContext, type ReactNode } from 'react';
import { sampleActivity, sampleBalanceCents, sampleCards } from '../data/sampleData';
import type { Card, CardStatus, Transaction } from '../types';
import { useStoredState } from './useStoredState';

type Wallet = {
  balanceCents: number;
  cards: Card[];
  activity: Transaction[];
  setCardStatus: (cardId: string, status: CardStatus) => void;
};

const WalletContext = createContext<Wallet | null>(null);

export function WalletProvider({ children }: { children: ReactNode }) {
  const [cards, setCards, loaded] = useStoredState('paydeck.cards', sampleCards);

  function setCardStatus(cardId: string, status: CardStatus) {
    setCards(cards.map((card) => (card.id === cardId ? { ...card, status } : card)));
  }

  if (!loaded) {
    return null;
  }

  return (
    <WalletContext.Provider
      value={{ balanceCents: sampleBalanceCents, cards, activity: sampleActivity, setCardStatus }}
    >
      {children}
    </WalletContext.Provider>
  );
}

export function useWallet(): Wallet {
  const wallet = useContext(WalletContext);
  if (wallet === null) {
    throw new Error('useWallet must be used inside <WalletProvider>');
  }
  return wallet;
}
```

`src/screens/CardDetailScreen.tsx`:

```tsx
import { Text } from 'react-native';
import { Button } from '../components/Button';
import { PaymentCard } from '../components/PaymentCard';
import { Screen } from '../components/Screen';
import { useNavigation } from '../navigation/NavigationContext';
import { useWallet } from '../state/WalletContext';

export function CardDetailScreen({ cardId }: { cardId: string }) {
  const { cards, setCardStatus } = useWallet();
  const { goBack } = useNavigation();
  const card = cards.find((c) => c.id === cardId);

  if (card === undefined) {
    return (
      <Screen title="Card">
        <Text>This card no longer exists.</Text>
      </Screen>
    );
  }

  const frozen = card.status === 'FROZEN';

  return (
    <Screen title={card.nickname}>
      <PaymentCard card={card} />
      <Button
        title={frozen ? 'Unfreeze card' : 'Freeze card'}
        variant="secondary"
        onPress={() => setCardStatus(card.id, frozen ? 'ACTIVE' : 'FROZEN')}
      />
      <Button
        title="Cancel card"
        variant="danger"
        onPress={() => {
          setCardStatus(card.id, 'CANCELLED');
          goBack();
        }}
      />
    </Screen>
  );
}
```

Things worth noticing:

- `WalletProvider` returns `null` until the saved cards have loaded, so no screen ever shows the sample data and then jumps.
- `useWallet` and `useNavigation` follow the same pattern: a context, a provider near the top, and a hook that fails clearly outside it.
- Only the cards are stored. The balance and activity are fixed sample data for now; top-ups arrive with the real API in Part 7.
- `formatSignedMoney` uses the real minus sign `−` (lesson 13.6).
